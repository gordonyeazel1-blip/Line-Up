<script>
  // --- Data Model ---
  let availablePlayers = [];
  let participatingPlayers = [];
  let lastGroupingMode = null; // 'foursomes' or 'threesomes'
  let lastGroupSize = null;

  function createPlayer(name, index) {
    return {
      id: crypto.randomUUID(),
      name: name.trim(),
      index: parseFloat(index),
      tier: 'D' // default, will be recalculated
    };
  }

  // --- Initial Seed from your list ---
  const initialData = [
    ["Dale Deselms", 23.1],
    ["Bill Moher", 19],
    ["Joe Tolitsky", 19.4],
    ["Mike Eldridge", 20],

    ["James Walker", 21.6],
    ["Chuck Louge", 16.7],
    ["Mike Tripp", 11.5],
    ["Jim Dugan", 23.5],
    ["Mickey Breznik", 16],
    ["Bob Merenes", 17],

    ["Eric Bernstein", 40.5],
    ["Ron Hearn", 27.6],
    ["Odie Odom", 22],
    ["Bill Baum", 22],
    ["Vinny Burke", 19.5],
    ["Steve Lauter", 25],
    ["Ricky Eldridge", 18.5],
    ["Joe Kaschac", 20.9],
    ["Jim Simpson", 13],
    ["Jim Smith", 20],
    ["Bill Healy", 32.8],
    ["Gordon YeazeL", 16.6],
    ["Drew Zwald", 22.5],
    ["Bill Austin", 8.3],
    ["Wayne McGirt", 17.8],
    ["Terry Britt", 14.7],
    ["Doug Wilson", 12.8],
    ["Gary Christian", 22.3],
    ["Bill Hapgood", 8.4],
    ["Bruce Berry", 13.2],
    ["Cole Pro", 15],
    ["Colby Wiener", 6.6],
    ["Frank Heacox", 17.5]
  ];

  availablePlayers = initialData.map(([name, index]) => createPlayer(name, index));
  recalcTiers();
  renderLists();

  // --- Tiering ---
  function recalcTiers() {
    const all = [...availablePlayers, ...participatingPlayers];
    all.sort((a, b) => a.index - b.index);

    all.forEach((p, i) => {
      if (i < 7) p.tier = 'A';
      else if (i < 14) p.tier = 'B';
      else if (i < 21) p.tier = 'C';
      else p.tier = 'D';
    });
  }

  // --- Rendering ---
  function renderLists() {
    recalcTiers();

    const availableList = document.getElementById('availableList');
    const participatingList = document.getElementById('participatingList');

    availableList.innerHTML = '';
    participatingList.innerHTML = '';

    // Sort by index ascending
    availablePlayers.sort((a, b) => a.index - b.index);
    participatingPlayers.sort((a, b) => a.index - b.index);

    availablePlayers.forEach(player => {
      const row = document.createElement('div');
      row.className = 'player-row';
      row.onclick = () => moveToParticipating(player.id);

      const name = document.createElement('div');
      name.className = 'player-name';
      name.textContent = player.name;

      const meta = document.createElement('div');
      meta.className = 'player-meta';
      meta.innerHTML = `
        <span>${player.index.toFixed(1)}</span>
        <span class="badge ${player.tier}">${player.tier}</span>
      `;

      row.appendChild(name);
      row.appendChild(meta);
      availableList.appendChild(row);
    });

    participatingPlayers.forEach(player => {
      const row = document.createElement('div');
      row.className = 'player-row';
      row.onclick = () => moveToAvailable(player.id);

      const name = document.createElement('div');
      name.className = 'player-name';
      name.textContent = player.name;

      const meta = document.createElement('div');
      meta.className = 'player-meta';
      meta.innerHTML = `
        <span>${player.index.toFixed(1)}</span>
        <span class="badge ${player.tier}">${player.tier}</span>
      `;

      row.appendChild(name);
      row.appendChild(meta);
      participatingList.appendChild(row);
    });
  }

  function moveToParticipating(id) {
    const idx = availablePlayers.findIndex(p => p.id === id);
    if (idx >= 0) {
      const [player] = availablePlayers.splice(idx, 1);
      participatingPlayers.push(player);
      renderLists();
      clearGroups();
    }
  }

  function moveToAvailable(id) {
    const idx = participatingPlayers.findIndex(p => p.id === id);
    if (idx >= 0) {
      const [player] = participatingPlayers.splice(idx, 1);
      availablePlayers.push(player);
      renderLists();
      clearGroups();
    }
  }

  // --- Clear buttons ---
  document.getElementById('clearAvailableBtn').onclick = () => {
    availablePlayers = [];
    renderLists();
    clearGroups();
  };

  document.getElementById('clearParticipatingBtn').onclick = () => {
    participatingPlayers = [];
    renderLists();
    clearGroups();
  };

  // --- Add Player ---
  document.getElementById('addPlayerBtn').onclick = () => {
    const name = document.getElementById('addName').value;
    const index = document.getElementById('addIndex').value;
    if (!name || !index) return;
    availablePlayers.push(createPlayer(name, index));
    document.getElementById('addName').value = '';
    document.getElementById('addIndex').value = '';
    renderLists();
    clearGroups();
  };

  // --- Paste List ---
  document.getElementById('pasteBtn').onclick = () => {
    const text = document.getElementById('pasteArea').value.trim();
    if (!text) return;
    const lines = text.split('\n');
    lines.forEach(line => {
      const parts = line.split(',');
      if (parts.length >= 2) {
        const name = parts[0].trim();
        const index = parseFloat(parts[1]);
        if (name && !isNaN(index)) {
          availablePlayers.push(createPlayer(name, index));
        }
      }
    });
    document.getElementById('pasteArea').value = '';
    renderLists();
    clearGroups();
  };

  // --- Grouping Logic ---
  function clearGroups() {
    document.getElementById('groupsContainer').innerHTML = '';
    document.getElementById('groupSummary').textContent = '';
    lastGroupingMode = null;
    lastGroupSize = null;
  }

  function createGroups(groupSize) {
    const players = [...participatingPlayers];
    if (players.length === 0) return;

    // Sort by index ascending
    players.sort((a, b) => a.index - b.index);

    const groupCount = Math.floor(players.length / groupSize);
    if (groupCount === 0) return;

    // Snake distribution
    const groups = Array.from({ length: groupCount }, () => []);
    let forward = true;
    let idx = 0;

    players.forEach(p => {
      groups[idx].push(p);
      if (forward) {
        idx++;
        if (idx >= groupCount) {
          idx = groupCount - 1;
          forward = false;
        }
      } else {
        idx--;
        if (idx < 0) {
          idx = 0;
          forward = true;
        }
      }
    });

    // Simple balancing by average index
    balanceGroupsByAverage(groups);

    renderGroups(groups, groupSize);
    lastGroupingMode = groupSize === 4 ? 'foursomes' : 'threesomes';
    lastGroupSize = groupSize;
  }

  function balanceGroupsByAverage(groups) {
    function avgIndex(group) {
      if (group.length === 0) return 0;
      return group.reduce((sum, p) => sum + p.index, 0) / group.length;
    }

    let changed = true;
    let safety = 0;
    while (changed && safety < 20) {
      safety++;
      changed = false;
      const avgs = groups.map(avgIndex);
      const maxIdx = avgs.indexOf(Math.max(...avgs));
      const minIdx = avgs.indexOf(Math.min(...avgs));
      if (maxIdx === minIdx) break;

      const strongGroup = groups[maxIdx];
      const weakGroup = groups[minIdx];

      const strongPlayer = strongGroup.reduce((acc, p) =>
        (!acc || p.index > acc.index) ? p : acc, null);
      const weakPlayer = weakGroup.reduce((acc, p) =>
        (!acc || p.index < acc.index) ? p : acc, null);

      if (!strongPlayer || !weakPlayer) break;

      const diffBefore = Math.abs(avgs[maxIdx] - avgs[minIdx]);

      const strongIdx = strongGroup.indexOf(strongPlayer);
      const weakIdx = weakGroup.indexOf(weakPlayer);
      strongGroup[strongIdx] = weakPlayer;
      weakGroup[weakIdx] = strongPlayer;

      const newAvgs = groups.map(avgIndex);
      const diffAfter = Math.abs(newAvgs[maxIdx] - newAvgs[minIdx]);

      if (diffAfter < diffBefore) {
        changed = true;
      } else {
        // revert
        strongGroup[strongIdx] = strongPlayer;
        weakGroup[weakIdx] = weakPlayer;
      }
    }
  }

  function renderGroups(groups, groupSize) {
    const container = document.getElementById('groupsContainer');
    const summary = document.getElementById('groupSummary');
    container.innerHTML = '';

    const totalPlayers = participatingPlayers.length;
    summary.textContent = `Created ${groups.length} group(s) of ${groupSize}. Total participating: ${totalPlayers}.`;

    groups.forEach((group, i) => {
      const card = document.createElement('div');
      card.className = 'group-card';

      const title = document.createElement('h3');
      title.textContent = `Group ${i + 1}`;
      card.appendChild(title);

      group.forEach(p => {
        const line = document.createElement('div');
        line.className = 'group-player';
        line.textContent = `${p.name} (${p.index.toFixed(1)}, ${p.tier})`;
        card.appendChild(line);
      });

      const avg = group.reduce((sum, p) => sum + p.index, 0) / group.length;
      const meta = document.createElement('div');
      meta.className = 'group-meta';
      meta.textContent = `Average index: ${avg.toFixed(1)}`;
      card.appendChild(meta);

      container.appendChild(card);
    });
  }

  // --- Buttons for grouping ---
  document.getElementById('foursomesBtn').onclick = () => {
    clearGroups();
    createGroups(4);
  };

  document.getElementById('threesomesBtn').onclick = () => {
    clearGroups();
    createGroups(3);
  };

  document.getElementById('threesomesOnlyBtn').onclick = () => {
    clearGroups();
    createGroups(3);
  };

  document.getElementById('reshuffleBtn').onclick = () => {
    if (!lastGroupSize) return;
    clearGroups();
    // Slight randomization before re-grouping
    participatingPlayers.sort(() => Math.random() - 0.5);
    createGroups(lastGroupSize);
  };
</script>
